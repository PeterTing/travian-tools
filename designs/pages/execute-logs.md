## 元件規格

### LogEntry

```typescript
interface LogEntryProps {
  log: ExecutionLog;
  onViewDetails: () => void;
  onRetry?: () => void;
}

interface ExecutionLog {
  id: string;
  timestamp: Date;
  type: 'build' | 'train' | 'upgrade';
  title: string;
  village: Village;
  status: 'success' | 'failed';
  duration: number;
  resourceCost: ResourceCost;
  error?: ExecutionError;
  details: ExecutionDetails;
}
```

### LogFilter

```typescript
interface LogFilterProps {
  filters: LogFilters;
  onChange: (filters: LogFilters) => void;
}

interface LogFilters {
  type: string | null;
  status: string | null;
  dateRange: '1d' | '7d' | '30d' | 'all';
  search: string;
}
```

### LogDetailModal

```typescript
interface LogDetailModalProps {
  log: ExecutionLog;
  isOpen: boolean;
  onClose: () => void;
  onRetry?: () => void;
}
```

### LogStats

```typescript
interface LogStatsProps {
  stats: {
    total: number;
    success: number;
    failed: number;
    totalResources: ResourceCost;
  };
  dateRange: string;
}
```

## API 整合

```typescript
// 取得執行日誌
GET /api/v1/logs?account_id={id}&type={type}&status={status}&date_range={range}&offset={offset}&limit={limit}
Response: {
  logs: ExecutionLog[];
  total: number;
  hasMore: boolean;
}

// 取得日誌統計
GET /api/v1/logs/stats?account_id={id}&date_range={range}
Response: LogStats

// 匯出日誌
GET /api/v1/logs/export?account_id={id}&format={csv|json}
Response: File

// 清除日誌
DELETE /api/v1/logs?account_id={id}&before={date}
Response: { deleted: number }
```
