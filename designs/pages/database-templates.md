## 元件規格

### TribeSelectorGrid

```typescript
interface TribeSelectorGridProps {
  value: Tribe;
  onChange: (tribe: Tribe) => void;
  showAll?: boolean;
}
```

### TemplateTypeSelector

```typescript
interface TemplateTypeSelectorProps {
  value: TemplateType;
  onChange: (type: TemplateType) => void;
}

type TemplateType = 'standard' | 'rush' | 'economy' | 'defense';
```

### TemplateOverview

```typescript
interface TemplateOverviewProps {
  template: Template;
  serverSpeed: number;
}

interface Template {
  id: string;
  tribe: Tribe;
  type: TemplateType;
  name: string;
  description: string;
  goals: TemplateGoal[];
  phases: TemplatePhase[];
  buildOrder: BuildStep[];
  resourceTips: string[];
}

interface TemplateGoal {
  building: string;
  level: number;
  icon: string;
}

interface TemplatePhase {
  number: number;
  startTime: number; // hours
  endTime: number;
  description: string;
}
```

### BuildOrderTable

```typescript
interface BuildOrderTableProps {
  steps: BuildStep[];
  serverSpeed: number;
  filter: 'all' | 'building' | 'resource' | 'troop';
  onStepClick?: (step: BuildStep) => void;
}

interface BuildStep {
  order: number;
  time: number; // seconds from start
  type: 'building' | 'resource' | 'troop';
  item: string;
  fromLevel: number;
  toLevel: number;
  cost: ResourceCost;
  note?: string;
}
```

### TemplateComparison

```typescript
interface TemplateComparisonProps {
  templates: Template[];
  onClose: () => void;
}
```

## API 整合

```typescript
// 取得所有模板
GET /api/v1/templates
Response: Template[]

// 取得特定種族模板
GET /api/v1/templates?tribe={tribe}
Response: Template[]

// 取得特定模板詳情
GET /api/v1/templates/{id}
Response: Template

// 取得模板建造順序
GET /api/v1/templates/{id}/build-order?speed={speed}
Response: BuildStep[]
```
