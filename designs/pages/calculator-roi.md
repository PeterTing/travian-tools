## 元件規格

### ResourceTypeSelector

```typescript
interface ResourceTypeSelectorProps {
  value: ResourceType;
  onChange: (type: ResourceType) => void;
}
```

### BonusSettings

```typescript
interface BonusSettingsProps {
  oasisBonus: number; // 0, 25, 50, 75, 150
  buildingBonus: number; // 0, 5, 10, 15, 20, 25
  plusAccount: boolean;
  onChange: (settings: BonusSettings) => void;
}
```

### ROIResult

```typescript
interface ROIResultProps {
  resourceType: ResourceType;
  currentLevel: number;
  upgradeCost: ResourceCost;
  productionBefore: number;
  productionAfter: number;
  bonusProduction: number;
  paybackTime: number; // hours
  thirtyDayROI: number; // percentage
}
```

```typescript
interface VillageResourceInputProps {
  woodFields: number[];
  clayFields: number[];
  ironFields: number[];

}
```

### ROIRankingTable

```typescript
interface ROIRankingTableProps {
  rankings: ROIRanking[];
  onSelect: (ranking: ROIRanking) => void;
}

interface ROIRanking {
  resourceType: ResourceType;
  fieldIndex: number;
  currentLevel: number;
  upgradeCost: number;
  paybackTime: number;
  recommendation: 'high' | 'medium' | 'low';
}
```

## 計算公式顯示

```
┌─────────────────────────────────────────────────────────────────────┐
│  📐 計算公式說明                                            [收起] │
├─────────────────────────────────────────────────────────────────────┤
│                                                                     │
│  基礎產量計算                                                        │
│  產量增加 = 新等級產量 - 當前等級產量                                 │
│                                                                     │
│  加成疊加規則                                                        │
│  最終產量 = 基礎產量 × (1 + 綠洲%) × (1 + 建築%) × (1 + Plus%)       │
│                                                                     │
│  ROI 計算                                                            │
│  回本時間(小時) = 升級總成本 ÷ 每小時產量增加                         │
│  30天ROI = (產量增加 × 24 × 30 - 升級成本) ÷ 升級成本 × 100%        │
│                                                                     │
└─────────────────────────────────────────────────────────────────────┘
```

## 狀態管理

```typescript
interface ROICalculatorState {
  mode: 'single' | 'village';
  // 單田模式
  resourceType: ResourceType;
  currentLevel: number;
  oasisBonus: number;
  buildingBonus: number;
  plusAccount: boolean;
  // 村莊模式
  villageId: string | null;
  manualFields: VillageFields | null;
  // 結果
  singleResult: ROIResult | null;
  villageRankings: ROIRanking[] | null;
  isCalculating: boolean;
}
```

## API 整合

```typescript
// 單田 ROI 計算
POST /api/v1/calculator/resource/roi
Request: {
  resource_type: string;
  current_level: number;
  oasis_bonus?: number;
  building_bonus?: number;
  plus_account?: boolean;
}
Response: ROIResult

// 村莊批量 ROI 分析
POST /api/v1/calculator/resource/village-roi
Request: {
  village_id?: string;
  fields?: VillageFields;
  oasis_bonus?: number;
  building_bonus?: number;
  plus_account?: boolean;
}
Response: {
  rankings: ROIRanking[];
  summary: {
    total_upgrade_cost: number;
    total_production_increase: number;
    average_payback_time: number;
  }
}
```
