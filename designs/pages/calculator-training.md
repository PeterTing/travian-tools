## API 整合

```typescript
POST /api/v1/calculator/training
Request: {
  tribe: string;
  troop_id: string;
  quantity: number;
  building_level: number;
  server_speed: number;
  use_great_barracks?: boolean;
}
Response: {
  unit_cost: ResourceCost;
  total_cost: ResourceCost;
  unit_time: number;
  total_time: number;
  crop_consumption: number;
  combat_stats: {
    attack: number;
    infantry_defense: number;
    cavalry_defense: number;
  };
  efficiency: {
    attack_per_crop: number;
    attack_per_cost: number;
  };
}
```
