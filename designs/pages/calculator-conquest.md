## 元件規格

### ConquestCalculator

```typescript
interface ConquestCalculatorProps {
  onCalculate: (params: ConquestParams) => void;
}

interface ConquestParams {
  targetPopulation: number;
  attackerTribe: Tribe;
  chiefCount: number;
  artefactBonus: number;
}
```

### ConquestResult

```typescript
interface ConquestResultProps {
  result: ConquestCalculation;
}

interface ConquestCalculation {
  loyaltyDropPerWave: {
    min: number;
    max: number;
    average: number;
  };
  wavesRequired: {
    best: number;
    average: number;
    worst: number;
  };
  trainingCost: ResourceCost;
  totalCost: ResourceCost;
  cropConsumption: number;
  trainingTime: number;
  simulation: WaveSimulation[];
}
```

## API 整合

