## 元件規格

### CropFieldsInput

農田等級輸入。

```typescript
interface CropFieldsInputProps {
  fields: number[]; // array of levels
  onChange: (fields: number[]) => void;
  fieldCount?: number; // default 6
}
```

### BonusInput

加成設定輸入。

```typescript
interface CropBonusInputProps {
  oasisBonus: number;
  flourMillLevel: number;
  bakeryLevel: number;
  plusAccount: boolean;
  onChange: (bonus: CropBonus) => void;
}
```

### TroopCropInput

部隊糧耗輸入。

```typescript
interface TroopCropInputProps {
  troops: TroopCrop[];
  onChange: (troops: TroopCrop[]) => void;
}

interface TroopCrop {
  troopId: string;
  count: number;
  cropConsumption: number; // per unit
}
```

### BalanceResult

平衡結果顯示。

```typescript
interface BalanceResultProps {
  production: number;
  consumption: number;
  balance: number;
  status: 'healthy' | 'warning' | 'critical';
  suggestions: Suggestion[];
  crisisInfo?: CrisisInfo;
}

interface CrisisInfo {
  hoursUntilEmpty: number;
  dailyLoss: number;
  troopLossPerHour: string;
}
```

## 狀態管理

```typescript
interface CropCalculatorState {
  dataSource: 'village' | 'manual';
  villageId: string | null;

  // 產量
  cropFields: number[];
  oasisBonus: number;
  flourMillLevel: number;
  bakeryLevel: number;
  plusAccount: boolean;

  // 消耗
  buildingPopulation: number;
  troops: TroopCrop[];

  // 結果
  result: CropBalanceResult | null;
  isCalculating: boolean;
}
```

## API 整合

```typescript
// 糧食平衡計算
POST /api/v1/calculator/crop/balance
Request: {
  village_id?: string;
  crop_fields?: number[];
  oasis_bonus?: number;
  flour_mill_level?: number;
  bakery_level?: number;
  plus_account?: boolean;
  building_population?: number;
  troops?: TroopCrop[];
}
Response: {
  production: number;
  consumption: number;
  balance: number;
  status: 'healthy' | 'warning' | 'critical';
  granary_capacity?: number;
  current_crop?: number;
  hours_until_empty?: number;
  suggestions: Suggestion[];
}
```
