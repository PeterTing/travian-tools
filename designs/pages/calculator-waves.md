## 元件規格

### WaveConfig

```typescript
interface WaveConfigProps {
  waves: Wave[];
  tribes: Tribe[];
  onChange: (waves: Wave[]) => void;
  onAddWave: () => void;
  onRemoveWave: (index: number) => void;
}

interface Wave {
  troopType: string;
  speed: number;
  purpose: WavePurpose;
}

type WavePurpose = 'scout' | 'clear_wall' | 'main_attack' | 'conquest' | 'raid' | 'other';
```

### WaveResultTable

```typescript
interface WaveResultTableProps {
  results: WaveResult[];
  targetTime: Date;
  interval: number;
}

interface WaveResult {
  waveNumber: number;
  troopType: string;
  sendTime: Date;
  arriveTime: Date;
  purpose: WavePurpose;
}
```

### TimelineVisualization

```typescript
interface TimelineVisualizationProps {
  results: WaveResult[];
  targetTime: Date;
}
```

## API 整合

```typescript
// 計算多波次發兵時間
POST /api/v1/calculator/waves
Request: {
  origin: Coordinate;
  destination: Coordinate;
  target_time: string; // ISO datetime
  interval_seconds: number;
  waves: Wave[];
  server_speed: number;
}
Response: {
  results: WaveResult[];
  warnings: string[];
}
```
