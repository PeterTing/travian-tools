## API 整合

```typescript
// 取得佇列項目
GET /api/v1/execute/queue?account_id={id}
Response: QueueItem[]

// 新增佇列項目
POST /api/v1/execute/queue
Request: {
  account_id: string;
  type: string;
  village_id: string;
  details: object;
}
Response: QueueItem

// 執行單一項目
POST /api/v1/execute/build
Request: {
  queue_item_id: string;
}
Response: {
  success: boolean;
  message: string;
  execution_time: number;
}

// 批量執行
POST /api/v1/execute/batch
Request: {
  queue_item_ids: string[];
}
Response: {
  results: ExecutionResult[];
}

// 刪除佇列項目
DELETE /api/v1/execute/queue/{id}
Response: { success: true }
```
