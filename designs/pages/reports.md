## 元件規格

### ReportCard

```typescript
interface ReportCardProps {
  report: Report;
  onClick: () => void;
  onSimulate?: () => void;
}

interface Report {
  id: string;
  type: ReportType;
  timestamp: Date;
  attacker: ReportParty;
  defender: ReportParty;
  result: ReportResult;
  loot?: ResourceCost;
  troops: TroopReport[];
  buildingDamage?: BuildingDamage[];
}

type ReportType = 'attack' | 'defense' | 'scout' | 'reinforcement';

interface ReportParty {
  playerName: string;
  villageName: string;
  coordinates: Coordinates;
  isOwn: boolean;
}

type ReportResult =
  | 'victory_total'      // 完全勝利
  | 'victory_partial'    // 部分勝利
  | 'defeat_total'       // 完全失敗
  | 'defeat_partial'     // 部分失敗
  | 'scout_success'      // 偵查成功
  | 'scout_partial'      // 部分偵查
  | 'scout_fail';        // 偵查失敗
```

interface ReportDetailProps {

## API 整合

```typescript
// 取得報告列表
GET /api/v1/reports?account_id={id}&type={type}&from={date}&to={date}&village_id={id}&page={page}&limit={limit}
Response: {
  reports: Report[];
  total: number;
  page: number;
  limit: number;
}

// 取得報告詳情
GET /api/v1/reports/{id}
Response: Report

// 匯入報告
POST /api/v1/reports/import
Request: { rawContent: string; accountId: string; }
Response: Report

// 刪除報告
DELETE /api/v1/reports/{id}
Response: { success: true }

// 取得報告統計
GET /api/v1/reports/statistics?account_id={id}&from={date}&to={date}
Response: ReportSummary
```
