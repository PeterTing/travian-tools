## 元件規格

### VillageCard

```typescript
interface VillageCardProps {
  village: Village;
  onViewDetails: () => void;
  onSync: () => void;
  onEdit: () => void;
}

interface Village {
  id: string;
  name: string;
  coordinates: { x: number; y: number };
  population: number;
  fieldDistribution: string; // "4-4-4-6"
  villageType: string;
  resources: ResourceStatus;
  cropBalance: number;
  troops: TroopUnit[];
  lastSync: Date;
}
```

### ResourceBar

```typescript
interface ResourceBarProps {
  type: ResourceType;
  current: number;
  capacity: number;
  production: number;
  showWarning?: boolean;
}
```

### VillageTabs

```typescript
interface VillageTabsProps {
  activeTab: 'resources' | 'buildings' | 'troops' | 'queue';
  onTabChange: (tab: string) => void;
  village: VillageDetail;
}
```

## API 整合

```typescript
// 取得村莊列表
GET /api/v1/villages
Response: Village[]

// 取得村莊詳情
GET /api/v1/villages/{id}
Response: VillageDetail

// 同步村莊數據
POST /api/v1/villages/{id}/sync
Response: Village

// 更新村莊（手動輸入）
PUT /api/v1/villages/{id}
Request: Partial<VillageDetail>
Response: VillageDetail
```
