interface CoordinateInputProps {
  value: Coordinate;
  onChange: (coord: Coordinate) => void;
  label: string;
  villages?: Village[];

interface Coordinate {
  x: number;
  y: number;
}

## API 整合

```typescript
// 計算距離
POST /api/v1/calculator/distance
Request: {
  origin: Coordinate;
  destination: Coordinate;
}
Response: {
  distance: number;
  formula: string;
}

// 計算行軍時間
POST /api/v1/calculator/travel-time
Request: {
  distance: number;
  tribe: string;
  server_speed: number;
}
Response: {
  travel_times: TravelTime[];
}

// 批量計算
POST /api/v1/calculator/distance/batch
Request: {
  origin: Coordinate;
  destinations: Coordinate[];
  tribe?: string;
  server_speed?: number;
}
Response: BatchResult[]
```
